    20→import { Label } from "@/components/ui/label";

    42→
    43→const summarySchema = z.object({
    44→  summary: z.string().min(20, "Summary must be at least 20 characters"),
    45→  headline: z.string().min(5, "Headline required"),
    46→  location: z.string().min(2, "Location required"),

    48→});
    49→type SummaryForm = z.infer<typeof summarySchema>;
    50→
    51→const SKILL_LEVELS = ["beginner", "intermediate", "advanced", "expert"] as const;
    52→type SkillLevel = (typeof SKILL_LEVELS)[number];
    53→
    54→export function CandidateProfile() {

    56→    () => candidateProfiles[0]
    57→  );
    58→  const [openToWork, setOpenToWork] = React.useState(profile.openToWork);
    59→  const [visibility, setVisibility] = React.useState(profile.visibility);
    60→  const [expDialog, setExpDialog] = React.useState(false);
    61→  const [eduDialog, setEduDialog] = React.useState(false);
    62→  const [editingExp, setEditingExp] = React.useState<WorkExperience | null>(null);
    63→  const [editingEdu, setEditingEdu] = React.useState<Education | null>(null);
    64→  const [newSkill, setNewSkill] = React.useState("");

    66→
    67→  const form = useForm<SummaryForm>({
    68→    resolver: zodResolver(summarySchema),
    69→    defaultValues: {
    70→      summary: profile.summary,
    71→      headline: profile.headline,
    72→      location: profile.location,
    73→      expectedSalary: profile.expectedSalary ?? 0,
    74→    },
    75→  });
    76→
    77→  // Compute completeness live from filled fields
    78→  const completeness = React.useMemo(() => computeCompleteness(profile), [profile]);
    79→
    80→  function onSummarySubmit(values: SummaryForm) {
    81→    setProfile((p) => ({ ...p, ...values, updatedAt: new Date().toISOString() }));
    82→    toast.success("Profile saved", { description: "Your headline, summary, and salary have been updated." });
    83→  }
    84→

    86→    setProfile((p) => {
    87→      const exists = p.experience.some((e) => e.id === exp.id);
    88→      const experience = exists
    89→        ? p.experience.map((e) => (e.id === exp.id ? exp : e))
    90→        : [...p.experience, exp];
    91→      return { ...p, experience };
    92→    });
    93→    setExpDialog(false);
    94→    setEditingExp(null);
    95→    toast.success(editingExp ? "Experience updated" : "Experience added");
    96→  }
    97→
    98→  function deleteExperience(id: string) {
    99→    setProfile((p) => ({ ...p, experience: p.experience.filter((e) => e.id !== id) }));
   100→    toast.success("Experience removed");
   101→  }
   102→
   103→  function saveEducation(edu: Education) {
   104→    setProfile((p) => {
   105→      const exists = p.education.some((e) => e.id === edu.id);
   106→      const education = exists
   107→        ? p.education.map((e) => (e.id === edu.id ? edu : e))
   108→        : [...p.education, edu];
   109→      return { ...p, education };
   110→    });
   111→    setEduDialog(false);
   112→    setEditingEdu(null);
   113→    toast.success(editingEdu ? "Education updated" : "Education added");
   114→  }
   115→
   116→  function deleteEducation(id: string) {
   117→    setProfile((p) => ({ ...p, education: p.education.filter((e) => e.id !== id) }));
   118→    toast.success("Education removed");
   119→  }
   120→
   121→  function addSkill() {
   122→    if (!newSkill.trim()) return;

   124→      toast.error("Skill already added");

   126→    }

   128→      ...p,
   129→      skills: [
   130→        ...p.skills,

   132→      ],
   133→    }));
   134→    setNewSkill("");
   135→    setNewSkillLevel("intermediate");

   137→  }
   138→
   139→  function updateSkillLevel(skill: string, level: SkillLevel) {
   140→    setProfile((p) => ({
   141→      ...p,
   142→      skills: p.skills.map((s) => (s.skill === skill ? { ...s, level } : s)),
   143→    }));
   144→  }
   145→
   146→  function removeSkill(skill: string) {
   147→    setProfile((p) => ({ ...p, skills: p.skills.filter((s) => s.skill !== skill) }));
   148→    toast.success("Skill removed");
   149→  }
   150→
   151→  function verifySkill(skill: string) {

   153→      description: `${skill} — we'll send you an assessment link shortly.`,
   154→    });
   155→  }

   163→          {
   164→            label: "Preview public view",
   165→            variant: "outline",
   166→            onClick: () => toast.info("Preview", { description: "Opening public profile preview..." }),
   167→          },

   177→              <h2 className="text-xl font-semibold">{profile.name}</h2>
   178→              <p className="text-sm text-muted-foreground">{profile.headline}</p>
   179→              <div className="mt-1 flex flex-wrap gap-x-3 gap-y-1 text-xs text-muted-foreground">
   180→                <span className="inline-flex items-center gap-1">
   181→                  <Icon name="MapPin" className="h-3 w-3" /> {profile.location}
   182→                </span>
   183→                <span className="inline-flex items-center gap-1">
   184→                  <Icon name="Briefcase" className="h-3 w-3" /> {profile.experienceYears} yrs exp
   185→                </span>
   186→                <span className="inline-flex items-center gap-1">
   187→                  <Icon name="Building2" className="h-3 w-3" /> {profile.currentCompany}
   188→                </span>
   189→              </div>
   190→            </div>

   232→            <Icon name="GraduationCap" className="h-4 w-4" /> Education
   233→          </TabsTrigger>
   234→          <TabsTrigger value="skills" className="gap-1">
   235→            <Icon name="Award" className="h-4 w-4" /> Skills

   238→            <Icon name="FolderGit2" className="h-4 w-4" /> Extras

   243→        </TabsList>

   245→        {/* Summary */}

   256→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   257→                <div className="space-y-1.5">
   258→                  <Label htmlFor="location">Location</Label>
   259→                  <Input id="location" {...form.register("location")} />
   260→                  {form.formState.errors.location && (
   261→                    <p className="text-xs text-destructive">{form.formState.errors.location.message}</p>

   263→                </div>

   275→              </div>

   278→                <Textarea id="summary" rows={4} {...form.register("summary")} />
   279→                {form.formState.errors.summary && (
   280→                  <p className="text-xs text-destructive">{form.formState.errors.summary.message}</p>
   281→                )}
   282→              </div>
   283→              <div className="flex justify-end">
   284→                <Button type="submit">Save changes</Button>

   290→        {/* Experience */}

   295→            action={
   296→              <Button
   297→                variant="outline"
   298→                size="sm"
   299→                onClick={() => {
   300→                  setEditingExp(null);
   301→                  setExpDialog(true);
   302→                }}
   303→              >
   304→                <Icon name="PenSquare" className="h-4 w-4" /> Add
   305→              </Button>
   306→            }
   307→          >
   308→            <div className="space-y-3">
   309→              {profile.experience.length === 0 ? (
   310→                <EmptyState
   311→                  icon={<Icon name="Briefcase" className="h-5 w-5" />}
   312→                  title="No experience added"
   313→                  description="Add your work history to build your profile."
   314→                />
   315→              ) : (
   316→                profile.experience.map((exp) => (
   317→                  <div key={exp.id} className="rounded-lg border p-4">
   318→                    <div className="flex items-start justify-between gap-2">
   319→                      <div className="min-w-0">
   320→                        <p className="font-medium">{exp.title}</p>
   321→                        <p className="text-sm text-muted-foreground">
   322→                          {exp.company}
   323→                          {exp.location ? ` · ${exp.location}` : ""}
   324→                        </p>
   325→                        <p className="text-xs text-muted-foreground">
   326→                          {fmtMonth(exp.startDate)} —{" "}
   327→                          {exp.current ? "Present" : exp.endDate ? fmtMonth(exp.endDate) : "—"}
   328→                        </p>
   329→                        {exp.description && (
   330→                          <p className="mt-2 text-sm text-muted-foreground">{exp.description}</p>
   331→                        )}
   332→                      </div>
   333→                      <div className="flex shrink-0 gap-1">
   334→                        <Button
   335→                          variant="ghost"
   336→                          size="icon"
   337→                          className="h-8 w-8"
   338→                          onClick={() => {
   339→                            setEditingExp(exp);
   340→                            setExpDialog(true);
   341→                          }}
   342→                        >
   343→                          <Icon name="PenSquare" className="h-4 w-4" />
   344→                        </Button>
   345→                        <Button
   346→                          variant="ghost"
   347→                          size="icon"
   348→                          className="h-8 w-8 text-destructive"
   349→                          onClick={() => deleteExperience(exp.id)}
   350→                        >
   351→                          <Icon name="AlertTriangle" className="h-4 w-4" />
   352→                        </Button>
   353→                      </div>
   354→                    </div>
   355→                  </div>
   356→                ))
   357→              )}
   358→            </div>
   359→          </SectionCard>
   360→        </TabsContent>
   361→
   362→        {/* Education */}
   363→        <TabsContent value="education">
   364→          <SectionCard
   365→            title="Education"
   366→            action={
   367→              <Button
   368→                variant="outline"
   369→                size="sm"
   370→                onClick={() => {
   371→                  setEditingEdu(null);
   372→                  setEduDialog(true);
   373→                }}
   374→              >
   375→                <Icon name="PenSquare" className="h-4 w-4" /> Add
   376→              </Button>
   377→            }
   378→          >
   379→            <div className="space-y-3">
   380→              {profile.education.length === 0 ? (
   381→                <EmptyState
   382→                  icon={<Icon name="GraduationCap" className="h-5 w-5" />}
   383→                  title="No education added"
   384→                />
   385→              ) : (
   386→                profile.education.map((edu) => (
   387→                  <div key={edu.id} className="rounded-lg border p-4">
   388→                    <div className="flex items-start justify-between gap-2">
   389→                      <div>
   390→                        <p className="font-medium">
   391→                          {edu.degree} · {edu.field}
   392→                        </p>
   393→                        <p className="text-sm text-muted-foreground">{edu.institution}</p>
   394→                        <p className="text-xs text-muted-foreground">
   395→                          {fmtMonth(edu.startDate)} — {edu.current ? "Present" : edu.endDate ? fmtMonth(edu.endDate) : "—"}
   396→                        </p>
   397→                        {edu.grade && (
   398→                          <Badge variant="secondary" className="mt-2">
   399→                            {edu.grade}
   400→                          </Badge>
   401→                        )}
   402→                      </div>
   403→                      <div className="flex shrink-0 gap-1">
   404→                        <Button
   405→                          variant="ghost"
   406→                          size="icon"
   407→                          className="h-8 w-8"
   408→                          onClick={() => {
   409→                            setEditingEdu(edu);
   410→                            setEduDialog(true);
   411→                          }}
   412→                        >
   413→                          <Icon name="PenSquare" className="h-4 w-4" />
   414→                        </Button>
   415→                        <Button
   416→                          variant="ghost"
   417→                          size="icon"
   418→                          className="h-8 w-8 text-destructive"
   419→                          onClick={() => deleteEducation(edu.id)}
   420→                        >
   421→                          <Icon name="AlertTriangle" className="h-4 w-4" />
   422→                        </Button>
   423→                      </div>
   424→                    </div>
   425→                  </div>
   426→                ))
   427→              )}
   428→            </div>
   429→          </SectionCard>
   430→        </TabsContent>
   431→
   432→        {/* Skills */}
   433→        <TabsContent value="skills">
   434→          <SectionCard title="Skills" description="Add skills and self-assess your level. Get verified to stand out.">
   435→            <div className="space-y-3">
   436→              {profile.skills.map((s) => (
   437→                <div
   438→                  key={s.skill}
   439→                  className="flex flex-col gap-2 rounded-lg border p-3 sm:flex-row sm:items-center sm:justify-between"
   440→                >
   441→                  <div className="flex items-center gap-2">
   442→                    <span className="font-medium">{s.skill}</span>

   476→                    </Select>

   513→              </div>
   514→            </div>
   515→          </SectionCard>
   516→        </TabsContent>
   517→
   518→        {/* Extras */}
   519→        <TabsContent value="extras">
   520→          <div className="grid grid-cols-1 gap-6 lg:grid-cols-2">
   521→            <SectionCard title="Certifications">
   522→              <div className="space-y-2">

   526→                    {c}
   527→                  </div>
   528→                ))}
   529→              </div>
   530→            </SectionCard>

   533→                {profile.languages.map((l) => (
   534→                  <div
   535→                    key={l.language}
   536→                    className="flex items-center justify-between rounded-lg border p-2 text-sm"
   537→                  >

   952→function computeCompleteness(p: CandidateProfile): number {
   953→  let filled = 0;
   954→  let total = 0;
   955→  const check = (cond: boolean) => {
   956→    total += 1;
   957→    if (cond) filled += 1;
   958→  };
   959→  check(!!p.headline);
   960→  check(p.summary.length > 30);
   961→  check(!!p.location);
   962→  check(p.experience.length >= 1);
   963→  check(p.education.length >= 1);
   964→  check(p.skills.length >= 5);
   965→  check(p.certifications.length >= 1);
   966→  check(p.languages.length >= 1);
   967→  check(p.achievements.length >= 1);
   968→  check(p.projects.length >= 1);
   969→  check(!!p.links.github || !!p.links.linkedin);
   970→  check(p.expectedSalary !== undefined && p.expectedSalary > 0);
   971→  return Math.round((filled / total) * 100);
   972→}
