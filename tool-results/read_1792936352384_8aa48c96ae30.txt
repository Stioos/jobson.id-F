   198→// Helper: navigate to a public route
   199→function useGo() {
   200→  const { navigate } = useNav();
   201→  return React.useCallback((view: string) => {
   202→    // "candidate/overview" and "employer/overview" cross-portal links
   203→    if (view.includes("/") && ["candidate", "employer", "recruiter", "training", "partner", "admin", "agency", "enterprise", "learner"].includes(view.split("/")[0])) {
   204→      const [portal, ...rest] = view.split("/");
   205→      navigate(portal as any, rest.join("/") || "overview");
   206→    } else {
   207→      navigate("public", view);
   208→    }
   209→  }, [navigate]);
   210→}
