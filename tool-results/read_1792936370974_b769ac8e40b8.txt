  1910→/**
  1911→ * Helper to resolve the exact program and subprogram track data.
  1912→ */
  1913→export function getProgramTrackData(
  1914→  mainProgramOrPreferred?: string | null,
  1915→  subProgramParam?: string | null
  1916→): ProgramTrackInfo | null {
  1917→  if (!mainProgramOrPreferred) return null;
  1918→
  1919→  let mainProg = mainProgramOrPreferred.trim();
  1920→  let subProg = subProgramParam ? subProgramParam.trim() : '';
  1921→
  1922→  // Handle strings like "Forward Deployed Engineer (FDE-FWD) — Data Engineer"
  1923→  if (mainProg.includes('—')) {
  1924→    const parts = mainProg.split('—').map((s) => s.trim());
  1925→    mainProg = parts[0];
  1926→    if (!subProg && parts[1]) {
  1927→      subProg = parts[1];
  1928→    }
  1929→  } else if (mainProg.includes(' - ')) {
  1930→    const parts = mainProg.split(' - ').map((s) => s.trim());
  1931→    mainProg = parts[0];
  1932→    if (!subProg && parts[1]) {
