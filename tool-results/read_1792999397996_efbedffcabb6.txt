  1023→/** Search all services across all categories */
  1024→export function searchAllServices(query: string) {
  1025→  if (!query.trim()) return [];
  1026→  const q = query.toLowerCase();
  1027→  const results: { service: ServiceItem; category: ServiceCategory }[] = [];
  1028→  for (const cat of serviceCategories) {
  1029→    for (const svc of cat.services) {
  1030→      if (
  1031→        svc.name.toLowerCase().includes(q) ||
  1032→        svc.shortDesc.toLowerCase().includes(q) ||
  1033→        cat.name.toLowerCase().includes(q)
  1034→      ) {
  1035→        results.push({ service: svc, category: cat });
  1036→      }
  1037→    }
  1038→  }
  1039→  return results;
  1040→}
